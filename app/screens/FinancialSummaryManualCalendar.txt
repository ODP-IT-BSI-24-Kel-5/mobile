// Define transaction type based on the API response
interface Transaction {
  id: string;
  transaction_number: string;
  amount: number;
  created_at: string;
  wallet: string;
  wallet_name: string;
  associate_wallet: string | null;
  associate_name: string | null;
  type: string; // "TOPUP" | "TRANSFER" | etc.
  notes: string | null;
  description: string;
  category: string | null;
  method: string | null;
  image_receipt: string | null;
}

// Define wallet type for selection
interface Wallet {
  id: string;
  name: string;
  balance: number;
  number: string;
  isMain: boolean;
}

// Define pagination metadata
interface PaginationMeta {
  current_page: number;
  page_size: number;
  total_items: number;
  total_pages: number;
  has_next: boolean;
  has_previous: boolean;
  first: boolean;
  last: boolean;
}

// Define category for filtering
interface Category {
  id: number;
  name: string;
  icon: string | null;
  sub_categories: SubCategory[];
  category: boolean;
  sub_category: boolean;
}

interface SubCategory {
  id: number;
  name: string;
  icon: string | null;
  sub_categories: any[];
  category: boolean;
  sub_category: boolean;
}

// Define transaction category summary
interface CategorySummary {
  name: string;
  amount: number;
  percentage: number;
  color: string;
  bgColor: string;
  icon: string;
  iconType: 'ionicons' | 'fontawesome';
  transactions: number;
}

export default function FinancialSummaryScreen() {
  const router = useRouter();
  const [loading, setLoading] = useState(true);
  const [transactions, setTransactions] = useState<Transaction[]>([]);
  const [filteredTransactions, setFilteredTransactions] = useState<Transaction[]>([]);
  const [wallets, setWallets] = useState<Wallet[]>([]);
  const [selectedWallet, setSelectedWallet] = useState<string | null>(null);
  const [categories, setCategories] = useState<Category[]>([]);
  const [expenseCategories, setExpenseCategories] = useState<SubCategory[]>([]);
  const [paginationMeta, setPaginationMeta] = useState<PaginationMeta | null>(null);
  const [currentPage, setCurrentPage] = useState(1);
  const [activeTab, setActiveTab] = useState<'income' | 'expense'>('expense');
  const [period, setPeriod] = useState(() => {
    const now = new Date();
    const startOfMonth = new Date(now.getFullYear(), now.getMonth(), 1); // First day of current month
    const endOfMonth = new Date(now.getFullYear(), now.getMonth() + 1, 0); // Last day of current month
    
    return {
      start: startOfMonth,
      end: endOfMonth,
    };
  });
  const [periodType, setPeriodType] = useState<'week' | 'month' | 'quarter'>('month');
  const [showWalletModal, setShowWalletModal] = useState(false);
  const [showPeriodModal, setShowPeriodModal] = useState(false);
  const [income, setIncome] = useState(0);
  const [expense, setExpense] = useState(0);
  const [loadingMore, setLoadingMore] = useState(false);
  const [categorySummaries, setCategorySummaries] = useState<CategorySummary[]>([]);

  // State for custom date selection
  const [showCustomDateModal, setShowCustomDateModal] = useState(false);
  const [tempPeriodStart, setTempPeriodStart] = useState<Date>(new Date());
  const [tempPeriodEnd, setTempPeriodEnd] = useState<Date>(new Date());
  const [customDateError, setCustomDateError] = useState<string | null>(null);
  const [datePickerMode, setDatePickerMode] = useState<'week' | 'month' | 'quarter'>('month');
  const [datePickerVisible, setDatePickerVisible] = useState(false);
  const [datePickerFor, setDatePickerFor] = useState<'start' | 'end'>('start');
  const [selectedYear, setSelectedYear] = useState(new Date().getFullYear());
  const [selectedMonthIndex, setSelectedMonthIndex] = useState(new Date().getMonth());

  // Month names array
  const monthNames = ['Januari', 'Februari', 'Maret', 'April', 'Mei', 'Juni', 
                      'Juli', 'Agustus', 'September', 'Oktober', 'November', 'Desember'];
  
  // Short month names for display
  const shortMonthNames = ['Jan', 'Feb', 'Mar', 'Apr', 'Mei', 'Jun', 'Jul', 'Agt', 'Sep', 'Okt', 'Nov', 'Des'];

  // Format date for display
  const formatDateForDisplay = (dateString: string) => {
    const date = new Date(dateString);
    return `${date.getDate()} ${shortMonthNames[date.getMonth()]} ${date.getFullYear()}`;
  };

  // Format period for display
  const formatPeriodForDisplay = () => {
    const startFormatted = `${period.start.getDate()} ${shortMonthNames[period.start.getMonth()]} ${period.start.getFullYear()}`;
    const endFormatted = `${period.end.getDate()} ${shortMonthNames[period.end.getMonth()]} ${period.end.getFullYear()}`;
    return `${startFormatted} - ${endFormatted}`;
  };

  // Get period type label
  const getPeriodTypeLabel = (): string => {
    switch (periodType) {
      case 'week':
        return 'Mingguan';
      case 'month':
        // Check if it's previous month
        const now = new Date();
        const currentMonth = now.getMonth();
        const periodMonth = period.start.getMonth();
        
        if (periodMonth !== currentMonth && period.end.getMonth() !== currentMonth) {
          return 'Bulan Lalu';
        }
        return 'Bulanan';
      case 'quarter':
        return 'Quarterly';
      default:
        return 'Periode';
    }
  };

  // Validate custom period selection
  const validateCustomPeriod = (): boolean => {
    // Reset error message
    setCustomDateError(null);
    
    const startTime = tempPeriodStart.getTime();
    const endTime = tempPeriodEnd.getTime();
    
    // Basic validation - start date must be before end date
    if (startTime > endTime) {
      setCustomDateError('Tanggal mulai harus sebelum tanggal akhir');
      return false;
    }
    
    // Specific validation based on mode
    switch (datePickerMode) {
      case 'week':
        // Calculate day difference
        const diffTime = Math.abs(endTime - startTime);
        const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24));
        
        if (diffDays !== 6) { // 7 days total (first day to seventh day)
          setCustomDateError('Rentang waktu harus tepat 7 hari');
          return false;
        }
        break;
        
      case 'quarter':
        // Calculate month difference
        const startYear = tempPeriodStart.getFullYear();
        const startMonth = tempPeriodStart.getMonth();
        const endYear = tempPeriodEnd.getFullYear();
        const endMonth = tempPeriodEnd.getMonth();
        
        const monthDiff = (endYear - startYear) * 12 + (endMonth - startMonth);
        
        if (monthDiff !== 2) { // 3 months total (first month to third month)
          setCustomDateError('Rentang waktu harus tepat 3 bulan');
          return false;
        }
        break;
    }
    
    return true;
  };

  // Apply custom period
  const applyCustomPeriod = () => {
    if (validateCustomPeriod()) {
      setPeriod({
        start: tempPeriodStart,
        end: tempPeriodEnd
      });
      
      setPeriodType(datePickerMode);
      setShowCustomDateModal(false);
    }
  };

  // Open custom date modal
  const openCustomDateModal = (mode: 'week' | 'month' | 'quarter') => {
    setDatePickerMode(mode);
    
    // Set default values based on mode
    const now = new Date();
    
    switch (mode) {
      case 'week':
        const weekStart = new Date(now);
        weekStart.setDate(now.getDate() - 6);
        setTempPeriodStart(weekStart);
        setTempPeriodEnd(now);
        break;
        
      case 'month':
        const monthStart = new Date(now.getFullYear(), now.getMonth(), 1);
        const monthEnd = new Date(now.getFullYear(), now.getMonth() + 1, 0);
        setTempPeriodStart(monthStart);
        setTempPeriodEnd(monthEnd);
        setSelectedMonthIndex(now.getMonth());
        setSelectedYear(now.getFullYear());
        break;
        
      case 'quarter':
        const quarterStart = new Date(now.getFullYear(), now.getMonth() - 2, 1);
        const quarterEnd = new Date(now.getFullYear(), now.getMonth() + 1, 0);
        setTempPeriodStart(quarterStart);
        setTempPeriodEnd(quarterEnd);
        break;
    }
    
    setShowCustomDateModal(true);
  };

  // Select month for month mode
  const selectMonth = (year: number, monthIndex: number) => {
    const monthStart = new Date(year, monthIndex, 1);
    const monthEnd = new Date(year, monthIndex + 1, 0);
    
    setTempPeriodStart(monthStart);
    setTempPeriodEnd(monthEnd);
    setSelectedMonthIndex(monthIndex);
    setSelectedYear(year);
  };

  // Update period by type
  const updatePeriodByType = (type: 'week' | 'month' | 'quarter') => {
    const now = new Date();
    
    switch (type) {
      case 'week':
        // Set period to last 7 days
        const weekAgo = new Date(now);
        weekAgo.setDate(now.getDate() - 6);
        setPeriod({
          start: weekAgo,
          end: now
        });
        break;
      
      case 'month':
        // Set period to beginning of month to end of month
        const startOfMonth = new Date(now.getFullYear(), now.getMonth(), 1);
        const endOfMonth = new Date(now.getFullYear(), now.getMonth() + 1, 0);
        setPeriod({
          start: startOfMonth,
          end: endOfMonth
        });
        break;
      
      case 'quarter':
        // Set period to 3 complete months
        const startOfQuarter = new Date(now);
        startOfQuarter.setMonth(now.getMonth() - 2);
        startOfQuarter.setDate(1); // Start of month
        
        const endOfQuarter = new Date(now.getFullYear(), now.getMonth() + 1, 0);
        setPeriod({
          start: startOfQuarter,
          end: endOfQuarter
        });
        break;
    }
  };

  useEffect(() => {
    // Initialize data
    checkAuth();
    fetchWallets();
    fetchCategories();
  }, []);

  useEffect(() => {
    // Fetch transactions when wallet or period changes
    if (selectedWallet) {
      console.log('Selected wallet changed, fetching transactions for:', selectedWallet);
      fetchTransactions(selectedWallet, 1);
    }
  }, [selectedWallet, period]);

  useEffect(() => {
    // Generate category summaries when transactions or active tab changes
    generateCategorySummaries();
  }, [activeTab, transactions, expenseCategories]);

  useEffect(() => {
    // Update period when period type changes
    updatePeriodByType(periodType);
  }, [periodType]);

  // Check user authentication
  const checkAuth = async () => {
    try {
      const token = await AsyncStorage.getItem('userToken');
      if (!token) {
        router.replace('/welcome');
      }
    } catch (error) {
      console.error('Error checking auth:', error);
      router.replace('/welcome');
    }
  };

  // Fetch user wallets
  const fetchWallets = async () => {
    try {
      const token = await AsyncStorage.getItem('userToken');
      if (!token) return;

      const response = await fetch('https://kelompok3.serverku.org/api/v1/users/wallets', {
        headers: {
          'Authorization': `Bearer ${token}`,
          'Content-Type': 'application/json'
        }
      });

      const data = await response.json();
      
      if (data.status === 'success' && data.wallets) {
        const mappedWallets = data.wallets.map((wallet: any, index: number) => ({
          id: wallet.number || index.toString(),
          name: wallet.name,
          balance: wallet.balance,
          number: wallet.number,
          isMain: wallet.is_main
        }));
        
        setWallets(mappedWallets);
        
        // Select the main wallet by default or the first one if no main wallet
        const mainWallet = mappedWallets.find((w: Wallet) => w.isMain);
        if (mainWallet && mainWallet.number) {
          setSelectedWallet(mainWallet.number);
        } else if (mappedWallets.length > 0 && mappedWallets[0].number) {
          setSelectedWallet(mappedWallets[0].number);
        }
      }
    } catch (error) {
      console.error('Error fetching wallets:', error);
    }
  };

  // Fetch transaction categories
  const fetchCategories = async () => {
    try {
      const token = await AsyncStorage.getItem('userToken');
      if (!token) return;

      const response = await fetch('https://kelompok3.serverku.org/api/v1/users/transactions/categories', {
        headers: {
          'Authorization': `Bearer ${token}`,
          'Content-Type': 'application/json'
        }
      });

      const data = await response.json();
      
      if (data.status === 'success' && data.transaction_categories) {
        setCategories(data.transaction_categories);
        
        // Extract Expenses subcategories
        const expensesCategory = data.transaction_categories.find(
          (cat: Category) => cat.name === 'Expenses'
        );
        
        if (expensesCategory && expensesCategory.sub_categories) {
          setExpenseCategories(expensesCategory.sub_categories);
        }
      }
    } catch (error) {
      console.error('Error fetching categories:', error);
    }
  };

  // Fetch transactions for a specific wallet
  const fetchTransactions = async (walletNumber: string, page: number) => {
    try {
      setLoading(true);
      const token = await AsyncStorage.getItem('userToken');
      if (!token) return;

      console.log(`Fetching transactions for wallet: ${walletNumber}, page: ${page}`);
      const response = await fetch(`https://kelompok3.serverku.org/api/v1/users/transactions/${walletNumber}?page=${page}&size=20`, {
        headers: {
          'Authorization': `Bearer ${token}`,
          'Content-Type': 'application/json'
        }
      });

      const data = await response.json();
      console.log(`Received ${data.data ? data.data.length : 0} transactions`);
      
      if (data.data && data.meta) {
        // Filter transactions to only include those for the selected wallet
        const filteredTransactions = data.data.filter((transaction: Transaction) => 
          transaction.wallet === walletNumber
        );
        
        console.log(`Filtered to ${filteredTransactions.length} transactions for wallet ${walletNumber}`);
        
        if (page === 1) {
          setTransactions(filteredTransactions);
          setFilteredTransactions(filteredTransactions);
        } else {
          setTransactions(prev => [...prev, ...filteredTransactions]);
          setFilteredTransactions(prev => [...prev, ...filteredTransactions]);
        }
        
        setPaginationMeta(data.meta);
        calculateFinancialSummary(page === 1 ? filteredTransactions : [...transactions, ...filteredTransactions]);
        setCurrentPage(page);
      }
    } catch (error) {
      console.error('Error fetching transactions:', error);
    } finally {
      setLoading(false);
      setLoadingMore(false);
    }
  };

  // Load more transactions
  const loadMoreTransactions = () => {
    if (paginationMeta?.has_next && selectedWallet) {
      setLoadingMore(true);
      fetchTransactions(selectedWallet, currentPage + 1);
    }
  };

  // Calculate income and expense from transactions
  const calculateFinancialSummary = (txns: Transaction[]) => {
    let totalIncome = 0;
    let totalExpense = 0;

    console.log(`Calculating summary for ${txns.length} transactions`);
    
    txns.forEach((transaction: Transaction) => {
      // TOPUP and receiving TRANSFER are considered income
      if (transaction.type === 'TOPUP' || 
          (transaction.type === 'TRANSFER' && transaction.description.includes('received'))) {
        totalIncome += transaction.amount;
      } 
      // Sending TRANSFER is considered expense
      else if (transaction.type === 'TRANSFER' && !transaction.description.includes('received')) {
        // Use Math.abs to make expense positive for the UI
        totalExpense += Math.abs(transaction.amount);
      }
    });

    console.log(`Calculated Income: ${totalIncome}, Expense: ${totalExpense}`);
    setIncome(totalIncome);
    setExpense(totalExpense);
  };

  // Find payment method styling by name (case insensitive)
  const findPaymentMethodByName = (name: string): {
    icon: string;
    iconType: 'ionicons' | 'fontawesome';
    iconColor: string;
    bgColor: string;
  } => {
    // Convert name to lowercase for case-insensitive comparison
    const lowercaseName = name.toLowerCase();
    
    for (const method of PAYMENT_METHODS) {
      if (method.name.toLowerCase() === lowercaseName) {
        return method as any;
      }
    }
    
    // Special handling for "Transfer In"
    if (lowercaseName.includes('transfer') && lowercaseName.includes('in')) {
      return PAYMENT_METHODS.find(m => m.name === 'Transfer In') || {
        icon: 'arrow-down-outline',
        iconType: 'ionicons',
        iconColor: '#3E9E8F',
        bgColor: '#E6F7F4'
      } as any;
    }
    
    // Default method styling
    return {
      icon: 'cash-outline',
      iconType: 'ionicons',
      iconColor: '#3E9E8F',
      bgColor: '#E6F7F4'
    };
  };

  // Find expense category by name or use default styling
  const findExpenseCategoryByName = (name: string): {
    id?: number;
    name: string;
    icon: string;
    iconType: 'ionicons' | 'fontawesome';
    color: string;
    bgColor: string;
  } => {
    const lowercaseName = name.toLowerCase();
    
    // Try to find category in expenseCategories
    const foundCategory = expenseCategories.find(cat => 
      cat.name.toLowerCase() === lowercaseName
    );
    
    if (foundCategory) {
      return {
        id: foundCategory.id,
        name: foundCategory.name,
        icon: getCategoryIcon(foundCategory.name),
        iconType: getCategoryIconType(foundCategory.name),
        color: getCategoryColor(foundCategory.name),
        bgColor: getCategoryBgColor(foundCategory.name)
      };
    }
    
    // Default styling based on name
    return {
      name,
      icon: getCategoryIcon(name),
      iconType: getCategoryIconType(name),
      color: getCategoryColor(name),
      bgColor: getCategoryBgColor(name)
    };
  };

  // Generate category summaries for pie chart
  const generateCategorySummaries = () => {
    // Skip if no transactions
    if (transactions.length === 0) {
      setCategorySummaries([]);
      return;
    }

    // Filter transactions based on active tab
    const relevantTransactions = transactions.filter(transaction => {
      if (activeTab === 'income') {
        return transaction.type === 'TOPUP' || 
              (transaction.type === 'TRANSFER' && transaction.description.includes('received'));
      } else { // expense
        return transaction.type === 'TRANSFER' && !transaction.description.includes('received');
      }
    });

    // Group by method or transaction type
    const categorized = new Map<string, { total: number, count: number, categoryId?: number }>();
    
    relevantTransactions.forEach(transaction => {
      let category = '';
      let categoryId: number | undefined = undefined;
      
      if (activeTab === 'income') {
        if (transaction.type === 'TOPUP') {
          category = transaction.method || 'Top-up';
        } else {
          category = 'Transfer In';
        }
      } else {
        // For expenses, use category from transaction if available
        if (transaction.category) {
          category = transaction.category;
        } else {
          // Try to find category by ID if available
          if (typeof transaction.category === 'string' && !isNaN(parseInt(transaction.category))) {
            const catId = parseInt(transaction.category);
            categoryId = catId;
            
            // Find category by ID
            const found = expenseCategories.find(cat => cat.id === catId);
            if (found) {
              category = found.name;
            } else {
              category = 'Transfer Out';
            }
          } else {
            category = 'Transfer Out';
          }
        }
      }
      
      if (categorized.has(category)) {
        const existing = categorized.get(category)!;
        categorized.set(category, {
          total: existing.total + Math.abs(transaction.amount),
          count: existing.count + 1,
          categoryId: existing.categoryId || categoryId
        });
      } else {
        categorized.set(category, {
          total: Math.abs(transaction.amount),
          count: 1,
          categoryId
        });
      }
    });

    // Calculate total for percentages
    const totalAmount = Array.from(categorized.values()).reduce((sum, item) => sum + item.total, 0);

    // Create summaries
    let summaries: CategorySummary[] = Array.from(categorized.entries()).map(([name, data]) => {
      if (activeTab === 'income') {
        // For income, use payment method styling
        const methodStyling = findPaymentMethodByName(name);
        return {
          name,
          amount: data.total,
          percentage: totalAmount > 0 ? Math.round((data.total / totalAmount) * 100) : 0,
          color: methodStyling.iconColor,
          bgColor: methodStyling.bgColor,
          icon: methodStyling.icon,
          iconType: methodStyling.iconType,
          transactions: data.count
        };
      } else {
        // For expense, use expense category styling
        const categoryStyling = findExpenseCategoryByName(name);
        
        return {
          name,
          amount: data.total,
          percentage: totalAmount > 0 ? Math.round((data.total / totalAmount) * 100) : 0,
          color: categoryStyling.color,
          bgColor: categoryStyling.bgColor,
          icon: categoryStyling.icon,
          iconType: categoryStyling.iconType,
          transactions: data.count
        };
      }
    });

    // Sort by amount (descending)
    summaries.sort((a, b) => b.amount - a.amount);
    
    setCategorySummaries(summaries);
  };

  // Format currency
  const formatCurrency = (value: number) => {
    return value.toLocaleString('id-ID');
  };

  // Render proper icon based on iconType
  const renderCategoryIcon = (category: CategorySummary) => {
    if (category.iconType === 'ionicons') {
      return <Ionicons name={category.icon as any} size={20} color={category.color} />;
    } else {
      return <FontAwesome5 name={category.icon} size={18} color={category.color} />;
    }
  };

  // Select a wallet
  const selectWallet = (wallet: Wallet) => {
    if (wallet.number) {
      console.log('Selecting wallet:', wallet.name, wallet.number);
      setSelectedWallet(wallet.number);
      setShowWalletModal(false);
    }
  };

  // Get selected wallet name
  const getSelectedWalletName = () => {
    if (!selectedWallet) return 'Semua tabungan & giro rupiah';
    const wallet = wallets.find(w => w.number === selectedWallet);
    return wallet ? wallet.name : 'Semua tabungan & giro rupiah';
  };

  // Create pie chart data
  const getPieChartData = () => {
    if (categorySummaries.length === 0) {
      return [
        {
          name: 'Tidak ada data',
          population: 1,
          color: '#E0E0E0',
          legendFontColor: '#7F7F7F',
          legendFontSize: 12
        }
      ];
    }

    return categorySummaries.map(category => ({
      name: category.name,
      population: category.amount,
      color: category.color,
      legendFontColor: '#7F7F7F',
      legendFontSize: 12
    }));
  };

  if (loading && transactions.length === 0) {
    return (
      <View className="flex-1 justify-center items-center bg-[#3E9E8F]">
        <ActivityIndicator size="large" color="#FFFFFF" />
        <Text className="mt-4 text-white">Loading data...</Text>
      </View>
    );
  }

  return (
    <SafeAreaView className="flex-1 bg-white">
      <StatusBar style="light" />
      <RNStatusBar backgroundColor="#3E9E8F" barStyle="light-content" />
      
      {/* Header */}
      <View className="pt-7 pb-4 px-4 bg-[#3E9E8F] mt-6">
        <View className="flex-row justify-between items-center">
          <View className="flex-row items-center">
            <TouchableOpacity onPress={() => router.back()}>
              <Ionicons name="arrow-back" size={24} color="#FFFFFF" />
            </TouchableOpacity>
            <Text className="ml-4 text-xl font-bold text-white">Rekap keuanganmu</Text>
          </View>
        </View>
        
        {/* Period Selector */}
        <TouchableOpacity 
          className="flex-row items-center justify-center mt-4"
          onPress={() => setShowPeriodModal(true)}
        >
          <Text className="text-white text-center">{getPeriodTypeLabel()}</Text>
          <View className="flex-row items-center justify-center">
            <Text className="text-white text-center font-bold mx-1">{formatPeriodForDisplay()}</Text>
            <Ionicons name="chevron-forward" size={16} color="#FFFFFF" />
          </View>
        </TouchableOpacity>
      </View>
      
      {/* Wallet Selector */}
      <TouchableOpacity 
        className="flex-row justify-between items-center mx-4 my-3 p-4 bg-white rounded-xl border border-gray-200"
        onPress={() => setShowWalletModal(true)}
      >
        <Text className="text-gray-800 font-medium">{getSelectedWalletName()}</Text>
        <Ionicons name="chevron-forward" size={18} color="#757575" />
      </TouchableOpacity>
      
      {/* Period Type Toggle */}
      <View className="mx-4 mb-3 px-1 py-1 bg-gray-200 rounded-full overflow-hidden">
        <View className="flex-row">
          <TouchableOpacity 
            className={`flex-1 py-2 px-1 ${periodType === 'week' ? 'bg-white rounded-full' : ''}`}
            onPress={() => setPeriodType('week')}
          >
            <Text className={`text-center font-medium text-xs ${periodType === 'week' ? 'text-[#3E9E8F]' : 'text-gray-700'}`}>
              Mingguan
            </Text>
          </TouchableOpacity>
          <TouchableOpacity 
            className={`flex-1 py-2 px-1 ${periodType === 'month' ? 'bg-white rounded-full' : ''}`}
            onPress={() => setPeriodType('month')}
          >
            <Text className={`text-center font-medium text-xs ${periodType === 'month' ? 'text-[#3E9E8F]' : 'text-gray-700'}`}>
              Bulanan
            </Text>
          </TouchableOpacity>
          <TouchableOpacity 
            className={`flex-1 py-2 px-1 ${periodType === 'quarter' ? 'bg-white rounded-full' : ''}`}
            onPress={() => setPeriodType('quarter')}
          >
            <Text className={`text-center font-medium text-xs ${periodType === 'quarter' ? 'text-[#3E9E8F]' : 'text-gray-700'}`}>
              Quarter
            </Text>
          </TouchableOpacity>
        </View>
        
        {/* Custom Date Selection Button */}
        <TouchableOpacity 
          className="mt-2 py-2 mx-1 bg-[#3E9E8F10] rounded-full flex-row justify-center items-center"
          onPress={() => openCustomDateModal(periodType)}
        >
          <Ionicons name="calendar-outline" size={14} color="#3E9E8F" />
          <Text className="text-[#3E9E8F] text-xs ml-1 font-medium">
            {periodType === 'week' ? 'Pilih Minggu Spesifik' : 
             periodType === 'month' ? 'Pilih Bulan Spesifik' : 
             'Pilih Quarter Spesifik'}
          </Text>
        </TouchableOpacity>
      </View>
      
      <ScrollView 
        className="flex-1 bg-gray-50"
        onScroll={({ nativeEvent }) => {
          const { layoutMeasurement, contentOffset, contentSize } = nativeEvent;
          const isCloseToBottom = layoutMeasurement.height + contentOffset.y >= contentSize.height - 20;
          
          if (isCloseToBottom && !loadingMore && paginationMeta?.has_next) {
            loadMoreTransactions();
          }
        }}
        scrollEventThrottle={400}
      >
        {/* Income/Expense Summary */}
        <View className="mx-4 my-3 p-4 bg-white rounded-xl shadow-sm">
          <View className="flex-row justify-between mb-4">
            <View>
              <View className="flex-row items-center">
                <View className="w-3 h-3 rounded-full bg-[#3E9E8F] mr-2" />
                <Text className="text-gray-600">Pemasukan</Text>
              </View>
              <Text className="text-lg font-bold mt-1">Rp{formatCurrency(income)}</Text>
            </View>
            <View className="border-l border-gray-200" style={{ width: 1 }} />
            <View>
              <View className="flex-row items-center">
                <View className="w-3 h-3 rounded-full bg-[#E57373] mr-2" />
                <Text className="text-gray-600">Pengeluaran</Text>
              </View>
              <Text className="text-lg font-bold mt-1">Rp{formatCurrency(expense)}</Text>
            </View>
          </View>
          
          <View className="border-t border-gray-100 pt-3">
            <Text className="text-gray-600 text-center">Selisih</Text>
            <Text className={`text-xl font-bold text-center ${income - expense >= 0 ? 'text-[#3E9E8F]' : 'text-[#E57373]'}`}>
              {income - expense >= 0 ? '+' : '-'}Rp{formatCurrency(Math.abs(income - expense))}
            </Text>
          </View>
        </View>
        
        {/* Income/Expense Toggle */}
        <View className="mx-4 mb-3 flex-row bg-gray-200 rounded-full overflow-hidden">
          <TouchableOpacity 
            className={`flex-1 py-3 px-4 ${activeTab === 'income' ? 'bg-white rounded-full' : ''}`}
            onPress={() => setActiveTab('income')}
          >
            <Text className={`text-center font-medium ${activeTab === 'income' ? 'text-[#3E9E8F]' : 'text-gray-700'}`}>
              Pemasukan
            </Text>
          </TouchableOpacity>
          <TouchableOpacity 
            className={`flex-1 py-3 px-4 ${activeTab === 'expense' ? 'bg-[#ECFF8C] rounded-full' : ''}`}
            onPress={() => setActiveTab('expense')}
          >
            <Text className={`text-center font-medium ${activeTab === 'expense' ? 'text-gray-800' : 'text-gray-700'}`}>
              Pengeluaran
            </Text>
          </TouchableOpacity>
        </View>
        
        {/* Category Breakdown Section */}
        <View className="mx-4 mb-4">
          <Text className="text-xl font-bold text-gray-800 mb-3">Rincian kategori</Text>
          
          {/* Pie Chart */}
          {categorySummaries.length > 0 ? (
            <View className="bg-white rounded-xl p-4 mb-3">
              <View className="items-center justify-center">
                {/* PieChart from react-native-chart-kit */}
                <PieChart
                  data={getPieChartData()}
                  width={screenWidth - 60}
                  height={220}
                  chartConfig={{
                    backgroundGradientFrom: "#FFFFFF",
                    backgroundGradientTo: "#FFFFFF",
                    color: (opacity = 1) => `rgba(0, 0, 0, ${opacity})`,
                  }}
                  accessor="population"
                  backgroundColor="transparent"
                  paddingLeft="15"
                  absolute
                  hasLegend={false}
                  center={[64,0]}
                />
                
                {/* Center Icon */}
                <View 
                  style={{ 
                    position: 'absolute', 
                    top: '50%', 
                    left: '50%', 
                    width: 60,
                    height: 60,
                    borderRadius: 30,
                    marginLeft: -30,
                    marginTop: -30,
                    backgroundColor: 'white',
                    justifyContent: 'center',
                    alignItems: 'center',
                    elevation: 5,
                    shadowColor: '#000',
                    shadowOffset: { width: 0, height: 2 },
                    shadowOpacity: 0.2,
                    shadowRadius: 3,
                  }}
                >
                  <Ionicons 
                    name={activeTab === 'income' ? 'arrow-down-outline' : 'arrow-up-outline'} 
                    size={36} 
                    color={activeTab === 'income' ? '#3E9E8F' : '#E57373'} 
                  />
                </View>
              </View>
            </View>
          ) : (
            <View className="bg-white rounded-xl p-8 mb-3 items-center justify-center">
              <Ionicons name="pie-chart-outline" size={48} color="#AEB5BB" />
              <Text className="text-gray-500 mt-3 text-center">
                Tidak ada data {activeTab === 'income' ? 'pemasukan' : 'pengeluaran'} untuk ditampilkan
              </Text>
            </View>
          )}
          
          {/* Category List */}
          <View className="bg-white rounded-xl overflow-hidden">
            {categorySummaries.map((category, index) => (
              <TouchableOpacity 
                key={index}
                className={`p-4 flex-row items-center justify-between ${
                  index !== categorySummaries.length - 1 ? 'border-b border-gray-100' : ''
                }`}
                onPress={() => {}}
              >
                <View className="flex-row items-center">
                  <View 
                    className="w-10 h-10 rounded-full items-center justify-center mr-3"
                    style={{ backgroundColor: category.bgColor }}
                  >
                    {renderCategoryIcon(category)}
                  </View>
                  <View>
                    <Text className="text-gray-800 font-medium">{category.name}</Text>
                    <Text className="text-gray-500 text-xs">
                      Dari {category.transactions} transaksi
                    </Text>
                  </View>
                </View>
                <View className="flex-row items-center">
                  <View>
                    <Text className="text-gray-800 text-right font-bold">
                      Rp{formatCurrency(category.amount)}
                    </Text>
                    <Text className="text-gray-600 text-xs text-right">
                      {category.percentage}%
                    </Text>
                  </View>
                  <Ionicons name="chevron-forward" size={18} color="#757575" className="ml-1" />
                </View>
              </TouchableOpacity>
            ))}
            
            {categorySummaries.length === 0 && (
              <View className="p-8 items-center">
                <Text className="text-gray-500">
                  Tidak ada data kategori untuk ditampilkan
                </Text>
              </View>
            )}
          </View>
          
          {/* Loading indicator for pagination */}
          {loadingMore && (
            <View className="py-4 items-center">
              <ActivityIndicator size="small" color="#3E9E8F" />
            </View>
          )}
        </View>
      </ScrollView>
      
      {/* Wallet Selection Modal */}
      <Modal
        visible={showWalletModal}
        transparent={true}
        animationType="fade"
        onRequestClose={() => setShowWalletModal(false)}
      >
        <View className="flex-1 bg-black/50 justify-center items-center p-4">
          <View className="bg-white rounded-xl w-full max-w-md p-4">
            <View className="flex-row justify-between items-center mb-4">
              <Text className="text-lg font-bold">Pilih Wallet</Text>
              <TouchableOpacity onPress={() => setShowWalletModal(false)}>
                <Ionicons name="close" size={24} color="#757575" />
              </TouchableOpacity>
            </View>
            
            <ScrollView className="max-h-60">
              {wallets.map((wallet) => (
                <TouchableOpacity 
                  key={wallet.id}
                  className="flex-row items-center p-3 border-b border-gray-100"
                  onPress={() => selectWallet(wallet)}
                >
                  <View className="w-10 h-10 rounded-full bg-[#3E9E8F] items-center justify-center mr-4">
                    <Text className="text-white font-bold">W</Text>
                  </View>
                  <View className="flex-1">
                    <Text className="text-gray-800 font-medium">{wallet.name}</Text>
                    <Text className="text-gray-500 text-xs">
                      Saldo: Rp{formatCurrency(wallet.balance || 0)}
                    </Text>
                  </View>
                  {selectedWallet === wallet.number && (
                    <Ionicons name="checkmark-circle" size={24} color="#3E9E8F" />
                  )}
                </TouchableOpacity>
              ))}
            </ScrollView>
          </View>
        </View>
      </Modal>
      {/* Period Selection Modal */}
      <Modal
        visible={showPeriodModal}
        transparent={true}
        animationType="fade"
        onRequestClose={() => setShowPeriodModal(false)}
      >
        <View className="flex-1 bg-black/50 justify-center items-center p-4">
          <View className="bg-white rounded-xl w-full max-w-md p-4">
            <View className="flex-row justify-between items-center mb-4">
              <Text className="text-lg font-bold">Pilih Periode</Text>
              <TouchableOpacity onPress={() => setShowPeriodModal(false)}>
                <Ionicons name="close" size={24} color="#757575" />
              </TouchableOpacity>
            </View>
            
            <ScrollView className="max-h-60">
              {/* Last 7 Days - Default Option */}
              <TouchableOpacity 
                className="p-3 border-b border-gray-100"
                onPress={() => {
                  const now = new Date();
                  const weekAgo = new Date(now);
                  weekAgo.setDate(now.getDate() - 6);
                  setPeriod({
                    start: weekAgo,
                    end: now
                  });
                  setPeriodType('week');
                  setShowPeriodModal(false);
                }}
              >
                <Text className="text-gray-800 font-medium">7 Hari Terakhir</Text>
                <Text className="text-gray-500 text-xs">
                  {(() => {
                    const now = new Date();
                    const weekAgo = new Date(now);
                    weekAgo.setDate(now.getDate() - 6);
                    return `${weekAgo.getDate()} ${shortMonthNames[weekAgo.getMonth()]} ${weekAgo.getFullYear()} - ${now.getDate()} ${shortMonthNames[now.getMonth()]} ${now.getFullYear()}`;
                  })()}
                </Text>
              </TouchableOpacity>

              {/* Custom Week - New Option */}
              <TouchableOpacity 
                className="p-3 border-b border-gray-100"
                onPress={() => openCustomDateModal('week')}
              >
                <Text className="text-gray-800 font-medium">Minggu Tertentu</Text>
                <Text className="text-gray-500 text-xs">
                  Pilih rentang 7 hari spesifik
                </Text>
              </TouchableOpacity>

              {/* Current Month */}
              <TouchableOpacity 
                className="p-3 border-b border-gray-100"
                onPress={() => {
                  const now = new Date();
                  const startOfMonth = new Date(now.getFullYear(), now.getMonth(), 1);
                  const endOfMonth = new Date(now.getFullYear(), now.getMonth() + 1, 0);
                  setPeriod({
                    start: startOfMonth,
                    end: endOfMonth
                  });
                  setPeriodType('month');
                  setShowPeriodModal(false);
                }}
              >
                <Text className="text-gray-800 font-medium">Bulan Ini</Text>
                <Text className="text-gray-500 text-xs">
                  {(() => {
                    const now = new Date();
                    const startOfMonth = new Date(now.getFullYear(), now.getMonth(), 1);
                    const endOfMonth = new Date(now.getFullYear(), now.getMonth() + 1, 0);
                    return `${startOfMonth.getDate()} ${shortMonthNames[startOfMonth.getMonth()]} ${startOfMonth.getFullYear()} - ${endOfMonth.getDate()} ${shortMonthNames[endOfMonth.getMonth()]} ${endOfMonth.getFullYear()}`;
                  })()}
                </Text>
              </TouchableOpacity>
              
              {/* Custom Month - New Option */}
              <TouchableOpacity 
                className="p-3 border-b border-gray-100"
                onPress={() => openCustomDateModal('month')}
              >
                <Text className="text-gray-800 font-medium">Bulan Tertentu</Text>
                <Text className="text-gray-500 text-xs">
                  Pilih bulan spesifik
                </Text>
              </TouchableOpacity>
              
              {/* Previous Month */}
              <TouchableOpacity 
                className="p-3 border-b border-gray-100"
                onPress={() => {
                  const now = new Date();
                  const startOfPrevMonth = new Date(now.getFullYear(), now.getMonth() - 1, 1);
                  const endOfPrevMonth = new Date(now.getFullYear(), now.getMonth(), 0);
                  setPeriod({
                    start: startOfPrevMonth,
                    end: endOfPrevMonth
                  });
                  setPeriodType('month');
                  setShowPeriodModal(false);
                }}
              >
                <Text className="text-gray-800 font-medium">Bulan Lalu</Text>
                <Text className="text-gray-500 text-xs">
                  {(() => {
                    const now = new Date();
                    const startOfPrevMonth = new Date(now.getFullYear(), now.getMonth() - 1, 1);
                    const endOfPrevMonth = new Date(now.getFullYear(), now.getMonth(), 0);
                    return `${startOfPrevMonth.getDate()} ${shortMonthNames[startOfPrevMonth.getMonth()]} ${startOfPrevMonth.getFullYear()} - ${endOfPrevMonth.getDate()} ${shortMonthNames[endOfPrevMonth.getMonth()]} ${endOfPrevMonth.getFullYear()}`;
                  })()}
                </Text>
              </TouchableOpacity>
              
              {/* Last 3 Months */}
              <TouchableOpacity 
                className="p-3 border-b border-gray-100"
                onPress={() => {
                  const now = new Date();
                  const threeMonthsAgo = new Date(now);
                  threeMonthsAgo.setMonth(now.getMonth() - 2);
                  threeMonthsAgo.setDate(1); // Start of month
                  setPeriod({
                    start: threeMonthsAgo,
                    end: now
                  });
                  setPeriodType('quarter');
                  setShowPeriodModal(false);
                }}
              >
                <Text className="text-gray-800 font-medium">3 Bulan Terakhir</Text>
                <Text className="text-gray-500 text-xs">
                  {(() => {
                    const now = new Date();
                    const threeMonthsAgo = new Date(now);
                    threeMonthsAgo.setMonth(now.getMonth() - 2);
                    threeMonthsAgo.setDate(1); // Start of month
                    return `${threeMonthsAgo.getDate()} ${shortMonthNames[threeMonthsAgo.getMonth()]} ${threeMonthsAgo.getFullYear()} - ${now.getDate()} ${shortMonthNames[now.getMonth()]} ${now.getFullYear()}`;
                  })()}
                </Text>
              </TouchableOpacity>

              {/* Custom Quarter - New Option */}
              <TouchableOpacity 
                className="p-3 border-b border-gray-100"
                onPress={() => openCustomDateModal('quarter')}
              >
                <Text className="text-gray-800 font-medium">Quarter Tertentu</Text>
                <Text className="text-gray-500 text-xs">
                  Pilih rentang 3 bulan spesifik
                </Text>
              </TouchableOpacity>
            </ScrollView>
          </View>
        </View>
      </Modal>
      
      {/* Custom Date Selection Modal */}
      <Modal
        visible={showCustomDateModal}
        transparent={true}
        animationType="fade"
        onRequestClose={() => setShowCustomDateModal(false)}
      >
        <View className="flex-1 bg-black/50 justify-center items-center p-4">
          <View className="bg-white rounded-xl w-full max-w-md p-4">
            <View className="flex-row justify-between items-center mb-4">
              <Text className="text-lg font-bold">
                {datePickerMode === 'week' ? 'Pilih Minggu' : 
                 datePickerMode === 'month' ? 'Pilih Bulan' : 'Pilih Quarter'}
              </Text>
              <TouchableOpacity onPress={() => setShowCustomDateModal(false)}>
                <Ionicons name="close" size={24} color="#757575" />
              </TouchableOpacity>
            </View>
            
            {/* Mode: Weekly */}
            {datePickerMode === 'week' && (
              <View>
                <Text className="text-gray-700 mb-2">Pilih tanggal awal minggu:</Text>
                <TouchableOpacity 
                  className="border border-gray-300 rounded-lg p-3 mb-4 flex-row justify-between items-center"
                  onPress={() => {
                    setDatePickerFor('start');
                    setDatePickerVisible(true);
                  }}
                >
                  <Text className="text-gray-800">{formatDateForDisplay(tempPeriodStart.toISOString())}</Text>
                  <Ionicons name="calendar-outline" size={20} color="#757575" />
                </TouchableOpacity>
                
                <Text className="text-gray-700 mb-2">Tanggal akhir minggu:</Text>
                <View className="border border-gray-300 rounded-lg p-3 mb-4 flex-row justify-between items-center bg-gray-100">
                  <Text className="text-gray-800">{formatDateForDisplay(tempPeriodEnd.toISOString())}</Text>
                  <Ionicons name="calendar-outline" size={20} color="#757575" />
                </View>
                
                <Text className="text-gray-500 text-xs mb-4">
                  Rentang waktu akan otomatis 7 hari dari tanggal awal yang dipilih.
                </Text>
                
                {customDateError && (
                  <Text className="text-red-500 mb-4">{customDateError}</Text>
                )}
                
                <TouchableOpacity 
                  className="bg-[#3E9E8F] py-3 rounded-lg"
                  onPress={() => {
                    // Set end date 6 days after start date (total 7 days)
                    const endDate = new Date(tempPeriodStart);
                    endDate.setDate(tempPeriodStart.getDate() + 6);
                    setTempPeriodEnd(endDate);
                    
                    // Apply the period
                    setPeriod({
                      start: tempPeriodStart,
                      end: endDate
                    });
                    setPeriodType('week');
                    setShowCustomDateModal(false);
                  }}
                >
                  <Text className="text-white text-center font-bold">Terapkan</Text>
                </TouchableOpacity>
              </View>
            )}
            
            {/* Mode: Monthly */}
            {datePickerMode === 'month' && (
              <View>
                <View className="flex-row justify-between items-center mb-4">
                  <TouchableOpacity onPress={() => setSelectedYear(selectedYear - 1)}>
                    <Ionicons name="chevron-back" size={24} color="#757575" />
                  </TouchableOpacity>
                  <Text className="text-gray-800 font-bold">{selectedYear}</Text>
                  <TouchableOpacity onPress={() => setSelectedYear(selectedYear + 1)}>
                    <Ionicons name="chevron-forward" size={24} color="#757575" />
                  </TouchableOpacity>
                </View>
                
                <View className="flex-row flex-wrap justify-between">
                  {monthNames.map((month, index) => (
                    <TouchableOpacity
                      key={index}
                      className={`w-[31%] py-3 px-2 mb-3 rounded-lg items-center ${
                        selectedMonthIndex === index ? 'bg-[#3E9E8F]' : 'bg-gray-100'
                      }`}
                      onPress={() => selectMonth(selectedYear, index)}
                    >
                      <Text className={`${
                        selectedMonthIndex === index ? 'text-white' : 'text-gray-800'
                      }`}>
                        {month.substring(0, 3)}
                      </Text>
                    </TouchableOpacity>
                  ))}
                </View>
                
                <View className="border-t border-gray-200 pt-4 mt-2 mb-4">
                  <Text className="text-gray-700">Periode dipilih:</Text>
                  <Text className="text-gray-800 font-bold">{`${
                    formatDateForDisplay(tempPeriodStart.toISOString())
                  } - ${
                    formatDateForDisplay(tempPeriodEnd.toISOString())
                  }`}</Text>
                </View>
                
                <TouchableOpacity 
                  className="bg-[#3E9E8F] py-3 rounded-lg"
                  onPress={() => {
                    setPeriod({
                      start: tempPeriodStart,
                      end: tempPeriodEnd
                    });
                    setPeriodType('month');
                    setShowCustomDateModal(false);
                  }}
                >
                  <Text className="text-white text-center font-bold">Terapkan</Text>
                </TouchableOpacity>
              </View>
            )}
            
            {/* Mode: Quarterly */}
            {datePickerMode === 'quarter' && (
              <View>
                <Text className="text-gray-700 mb-2">Pilih bulan awal quarter:</Text>
                <TouchableOpacity 
                  className="border border-gray-300 rounded-lg p-3 mb-4 flex-row justify-between items-center"
                  onPress={() => {
                    setDatePickerFor('start');
                    setDatePickerVisible(true);
                  }}
                >
                  <Text className="text-gray-800">{monthNames[tempPeriodStart.getMonth()]} {tempPeriodStart.getFullYear()}</Text>
                  <Ionicons name="calendar-outline" size={20} color="#757575" />
                </TouchableOpacity>
                
                <Text className="text-gray-700 mb-2">Bulan akhir quarter:</Text>
                <View className="border border-gray-300 rounded-lg p-3 mb-4 flex-row justify-between items-center bg-gray-100">
                  <Text className="text-gray-800">{monthNames[tempPeriodEnd.getMonth()]} {tempPeriodEnd.getFullYear()}</Text>
                  <Ionicons name="calendar-outline" size={20} color="#757575" />
                </View>
                
                <Text className="text-gray-500 text-xs mb-4">
                  Rentang waktu akan otomatis 3 bulan dari bulan awal yang dipilih.
                </Text>
                
                {customDateError && (
                  <Text className="text-red-500 mb-4">{customDateError}</Text>
                )}
                
                <TouchableOpacity 
                  className="bg-[#3E9E8F] py-3 rounded-lg"
                  onPress={() => {
                    // Set start to beginning of selected month
                    const startDate = new Date(tempPeriodStart);
                    startDate.setDate(1); // Set to first day of month
                    
                    // Set end to end of month that is 2 months after start (total 3 months)
                    const endDate = new Date(tempPeriodStart);
                    endDate.setMonth(tempPeriodStart.getMonth() + 2);
                    endDate.setDate(new Date(endDate.getFullYear(), endDate.getMonth() + 1, 0).getDate()); // Set to last day of month
                    
                    setTempPeriodEnd(endDate);
                    
                    // Apply the period
                    setPeriod({
                      start: startDate,
                      end: endDate
                    });
                    setPeriodType('quarter');
                    setShowCustomDateModal(false);
                  }}
                >
                  <Text className="text-white text-center font-bold">Terapkan</Text>
                </TouchableOpacity>
              </View>
            )}
          </View>
        </View>
      </Modal>
      
      {/* Date Picker for Week and Quarter mode */}
      {datePickerVisible && (datePickerMode === 'week' || datePickerMode === 'quarter') && (
        <Modal
          visible={datePickerVisible}
          transparent={true}
          animationType="slide"
        >
          <View className="flex-1 bg-black/50 justify-end">
            <View className="bg-white p-4 rounded-t-xl">
              <View className="flex-row justify-between items-center mb-4">
                <TouchableOpacity onPress={() => setDatePickerVisible(false)}>
                  <Text className="text-[#3E9E8F] font-medium">Batal</Text>
                </TouchableOpacity>
                <Text className="text-gray-800 font-bold">Pilih Tanggal</Text>
                <TouchableOpacity 
                  onPress={() => {
                    if (datePickerMode === 'week') {
                      // For week mode, just set start date, end will be calculated automatically
                      if (datePickerFor === 'start') {
                        // Update end date automatically to 6 days after
                        const endDate = new Date(tempPeriodStart);
                        endDate.setDate(tempPeriodStart.getDate() + 6);
                        setTempPeriodEnd(endDate);
                      }
                    } else if (datePickerMode === 'quarter') {
                      // For quarter mode, set start month, end will be calculated automatically
                      if (datePickerFor === 'start') {
                        const startDate = new Date(tempPeriodStart);
                        startDate.setDate(1); // Set to first day of month
                        
                        const endDate = new Date(tempPeriodStart);
                        endDate.setMonth(tempPeriodStart.getMonth() + 2);
                        endDate.setDate(new Date(endDate.getFullYear(), endDate.getMonth() + 1, 0).getDate()); // Set to last day of month
                        
                        setTempPeriodEnd(endDate);
                      }
                    }
                    
                    setDatePickerVisible(false);
                  }}
                >
                  <Text className="text-[#3E9E8F] font-medium">Selesai</Text>
                </TouchableOpacity>
              </View>
              
              {/* Here we need a DateTimePicker component that is platform-specific */}
              {/* For simplicity, let's create a custom date picker UI for Android */}
              
              <View className="py-4">
                <Text className="text-center text-gray-700 mb-2">
                  {datePickerMode === 'week' ? 'Pilih tanggal awal minggu:' : 'Pilih bulan awal quarter:'}
                </Text>
                
                {/* Year picker */}
                <View className="flex-row justify-center items-center mb-4">
                  <TouchableOpacity onPress={() => {
                    if (datePickerFor === 'start') {
                      const newDate = new Date(tempPeriodStart);
                      newDate.setFullYear(newDate.getFullYear() - 1);
                      setTempPeriodStart(newDate);
                    }
                  }}>
                    <Ionicons name="chevron-back" size={24} color="#757575" />
                  </TouchableOpacity>
                  <Text className="mx-4 text-lg font-bold">
                    {datePickerFor === 'start' ? tempPeriodStart.getFullYear() : tempPeriodEnd.getFullYear()}
                  </Text>
                  <TouchableOpacity onPress={() => {
                    if (datePickerFor === 'start') {
                      const newDate = new Date(tempPeriodStart);
                      newDate.setFullYear(newDate.getFullYear() + 1);
                      setTempPeriodStart(newDate);
                    }
                  }}>
                    <Ionicons name="chevron-forward" size={24} color="#757575" />
                  </TouchableOpacity>
                </View>
                
                {/* Month picker for quarter mode */}
                {datePickerMode === 'quarter' && (
                  <View className="flex-row flex-wrap justify-between mb-4">
                    {monthNames.map((month, index) => (
                      <TouchableOpacity
                        key={index}
                        className={`w-[31%] py-3 px-2 mb-3 rounded-lg items-center ${
                          datePickerFor === 'start' && tempPeriodStart.getMonth() === index ? 'bg-[#3E9E8F]' : 'bg-gray-100'
                        }`}
                        onPress={() => {
                          if (datePickerFor === 'start') {
                            const newDate = new Date(tempPeriodStart);
                            newDate.setMonth(index);
                            newDate.setDate(1); // Set to first day of month
                            setTempPeriodStart(newDate);
                          }
                        }}
                      >
                        <Text className={`${
                          datePickerFor === 'start' && tempPeriodStart.getMonth() === index ? 'text-white' : 'text-gray-800'
                        }`}>
                          {month.substring(0, 3)}
                        </Text>
                      </TouchableOpacity>
                    ))}
                  </View>
                )}
                
                {/* Date picker for week mode */}
                {datePickerMode === 'week' && (
                  <View>
                    {/* Month picker for week mode */}
                    <View className="flex-row justify-center items-center mb-4">
                      <TouchableOpacity onPress={() => {
                        if (datePickerFor === 'start') {
                          const newDate = new Date(tempPeriodStart);
                          newDate.setMonth(newDate.getMonth() - 1);
                          setTempPeriodStart(newDate);
                        }
                      }}>
                        <Ionicons name="chevron-back" size={24} color="#757575" />
                      </TouchableOpacity>
                      <Text className="mx-4 text-lg font-medium">
                        {monthNames[datePickerFor === 'start' ? tempPeriodStart.getMonth() : tempPeriodEnd.getMonth()]}
                      </Text>
                      <TouchableOpacity onPress={() => {
                        if (datePickerFor === 'start') {
                          const newDate = new Date(tempPeriodStart);
                          newDate.setMonth(newDate.getMonth() + 1);
                          setTempPeriodStart(newDate);
                        }
                      }}>
                        <Ionicons name="chevron-forward" size={24} color="#757575" />
                      </TouchableOpacity>
                    </View>
                    
                    {/* Calendar grid for week */}
                    <View className="flex-row flex-wrap justify-between">
                      {['Min', 'Sen', 'Sel', 'Rab', 'Kam', 'Jum', 'Sab'].map((day, index) => (
                        <View key={`header-${index}`} className="w-[14%] py-2 items-center">
                          <Text className="text-gray-600 text-xs">{day}</Text>
                        </View>
                      ))}
                      
                      {(() => {
                        const date = datePickerFor === 'start' ? tempPeriodStart : tempPeriodEnd;
                        const month = date.getMonth();
                        const year = date.getFullYear();
                        
                        // First day of the selected month
                        const firstDay = new Date(year, month, 1);
                        // Day of week for the first day (0 = Sunday, 6 = Saturday)
                        const startingDay = firstDay.getDay();
                        // Number of days in month
                        const daysInMonth = new Date(year, month + 1, 0).getDate();
                        
                        // Generate calendar grid
                        const days = [];
                        
                        // Empty cells for days before the 1st
                        for (let i = 0; i < startingDay; i++) {
                          days.push(
                            <View key={`empty-${i}`} className="w-[14%] py-2 items-center">
                              <Text className="text-gray-300">-</Text>
                            </View>
                          );
                        }
                        
                        // Cells for dates in month
                        for (let i = 1; i <= daysInMonth; i++) {
                          const thisDate = new Date(year, month, i);
                          const isSelected = datePickerFor === 'start' ? 
                            thisDate.getDate() === tempPeriodStart.getDate() :
                            thisDate.getDate() === tempPeriodEnd.getDate();
                          
                          days.push(
                            <TouchableOpacity 
                              key={`day-${i}`} 
                              className={`w-[14%] py-2 items-center`}
                              onPress={() => {
                                if (datePickerFor === 'start') {
                                  const newDate = new Date(year, month, i);
                                  setTempPeriodStart(newDate);
                                }
                              }}
                            >
                              <View className={`w-8 h-8 rounded-full items-center justify-center ${
                                isSelected ? 'bg-[#3E9E8F]' : 'bg-transparent'
                              }`}>
                                <Text className={`${isSelected ? 'text-white' : 'text-gray-800'}`}>
                                  {i}
                                </Text>
                              </View>
                            </TouchableOpacity>
                          );
                        }
                        
                        return days;
                      })()}
                    </View>
                  </View>
                )}
              </View>
            </View>
          </View>
        </Modal>
      )}
    </SafeAreaView>
  );
}
